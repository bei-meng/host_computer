# 注释使用:#或者;
# 立即数只支持：16进制数格式0x10或者0X10, 10进制数格式10101,10进制数最高位不能为0
# 指令名称后面为i的表示对立即数或常量的操作
# 立即数大小不能超过256，因为只有8位,可以通过移位乘法来实现
# 一定要记得reset：cim_reset
# 指令不区分大小写


# 这两条指令不用管
const_i imm,10                                          # 注册常量imm, 值为10,指令中的立即数可以使用
# free_reg reg1                                           # 释放变量reg1所占寄存器(如果后面没有用到这个寄存器的值)

const_i imm0,10
const_i imm1,10
const_i mode,1
mov_i reg0,1
mov_i reg1,2
mov_i reg2,3

# 指令模式使用的指令,不涉及寄存器操作
label:

jmp label                                               # 跳转到对应标签对应的IR_ram_addr
set_row_bank_and_data_i imm1,imm0                       # 从din_ram配置行bank,imm1为row_bank_mask(8bit),din_ram[imm0]里面是row_index_mask(32bit)
set_col_bank_and_data_i imm1,imm0                       # 从din_ram配置列bank,imm1为col_bank_mask(8bit),din_ram[imm0]里面是col_index_mask(32bit)
set_daci imm1,imm0                                      # 设置dac的第imm1[0:11]个通道的电压码值(16bit)为imm0
row_read_rram_32ch_to_dout_i imm                        # 从行读16路TIA,结果存储在dout_ram[imm](每个数据单元256bit)
col_read_rram_32ch_to_dout_i imm                        # 从列读16路TIA,结果存储在dout_ram[imm](每个数据单元256bit)
row_write_pulse                                         # 从行写,产生row写pulse
col_write_pulse                                         # 从列写,产生col写pulse
cim_reset                                               # 清零latch

# 涉及寄存器
bge_r reg1,reg0,label                                   # if reg1 >= reg0 就跳转到对应标签处
add_i reg1,reg0,imm                                     # reg1 = reg0 + imm
exit                                                    # 结束指令
load_din_ram_to_reg reg1,reg0                           # 从din_ram取数据,reg1 = din_ram[reg0]
add_r reg2,reg1,reg0                                    # reg2 = reg1 + reg0
sub_r reg2,reg1,reg0                                    # reg2 = reg1 - reg0
xor_i reg1,reg0,imm                                     # reg1 = reg0 ^ imm
sll_r reg2,reg1,reg0                                    # reg2 = reg1 << reg0
srl_r reg2,reg1,reg0                                    # reg2 = reg1 >> reg0

set_row_bank_and_data_r reg1,reg0                       # 配置行bank,reg1 为row_bank_mask,reg0 为row_index_mask
set_col_bank_and_data_r reg1,reg0                       # 配置列bank,reg1 为col_bank_mask,reg0 为col_index_mask

row_read_rram_1ch_to_dout reg2,reg1,reg0                # 从行读,reg2 为tia的掩码,reg1 为结果存储的dout_ram地址,reg0 为哪一块dout_ram(每个数据单元16bit)
col_read_rram_1ch_to_dout reg2,reg1,reg0                # 从列读,reg2 为tia的掩码,reg1 为结果存储的dout_ram地址,reg0 为哪一块dout_ram(每个数据单元16bit)

return_dout reg2,reg1,reg0                              # 数据返回上位机,reg2 为要返回的数据长度,reg1 为dout_ram读取的起始地址,reg0 为哪一块dout_ram(每个数据单元16bit)

row_read_rram_1ch_to_reg reg1,reg0                      # 从行读,reg1 为tia的掩码,tia读出的结果存在reg0
col_read_rram_1ch_to_reg reg1,reg0                      # 从列读,reg1 为tia的掩码,tia读出的结果存在reg0

row_ctrl_i imm0                                         # 设置row_ctrl为imm0
col_ctrl_i imm0                                         # 设置col_ctrl为imm0
row_col_sw_i imm0                                       # 设置row_col_sw为imm0

# pl新增指令
row_read_rram_32ch_to_diff
col_read_rram_32ch_to_diff

cal_ch_mask imm1,imm0

apu_calc mode
    apu_calc_relu                                       # apu_calc 0
    apu_calc_sigmoid                                    # apu_calc 1
    apu_calc_tanh                                       # apu_calc 2


element_wise reg0,imm0,mode
    set_elewise_APU_2_mul_A                             # element_wise 0,0,0
    set_elewise_actv_2_mul_A                            # element_wise 0,0,1
    set_elewise_acc_2_mul_A                             # element_wise 0,0,2
    set_elewise_APU_2__mul_B                            # element_wise 0,0,3
    set_elewise_actv_2_mul_B                            # element_wise 0,0,4
    set_elewise_reg_2_mul_B reg2                        # element_wise reg2,0,5
    set_elewise_reg_2_shifter0 reg2                     # element_wise reg2,0,6
    set_elewise_mul_shifter0_calc                       # element_wise 0,0,7
    set_elewise_mul_2_adder_A                           # element_wise 0,0,8
    set_elewise_acc_2_adder_A                           # element_wise 0,0,9
    set_elewise_mul_2_adder_B                           # element_wise 0,0,A
    set_elewise_actv_2_adder_B                          # element_wise 0,0,B
    set_elewise_adder_shifter1_calc                     # element_wise 0,0,C
    set_elewise_adder_clear                             # element_wise 0,0,D
    set_elewise_fast                                    # element_wise 0,0,E
    set_elewise_reg_2_shifter1 reg2                     # element_wise reg2,0,F


set_actv_ram_addr reg0
load_actv_ram
store_actv_ram mode
    store_actv_ram_shifter0                             # store_actv_ram 0
    store_actv_ram_shifter1                             # store_actv_ram 1
    store_actv_ram_apu                                  # store_actv_ram 2


reshape_buffer_clear
set_reshape_buffer_addr reg0
store_reshape_buffer
load_reshape_buffer_t reg2,reg1,reg0

set_adc_discard_nbits imm


IR_ram_addr:
# RRAM基础控制
MOV_i reg,imm                                           # reg = imm
MOV_r reg2,reg1                                         # reg2 = reg1
CMP_i reg,imm                                           # reg和imm比较,结果返回比较状态寄存器
CMP_r reg2,reg1                                         # reg1和reg1比较,结果返回比较状态寄存器
JEQ IR_ram_addr
JNE IR_ram_addr
JLT IR_ram_addr
JGT IR_ram_addr
JLE IR_ram_addr
JGE IR_ram_addr
